    41→  let heart = (
    42→    (0,1,1,0,0,1,1,0),

    52→    let out = ()
    53→    for r in range(8) { for c in range(8) {
    54→      out.push(if heart.at(r).at(c) == 1 { rect(width: cell, height: cell, fill: tone, radius: 0pt, stroke: none) } else { [] })
    55→    } }
    56→    out
    57→  }

   570→// pie chart built from thin polygon fans (accurate wedges, no external deps)
   571→#let piechart-example(values, labels, colors, pw: 46mm) = box(width: 100%, {
   572→  let total = values.sum()
   573→  let r = 17.5mm
   574→  let cx = r + 0.5mm
   575→  let cy = r + 0.5mm
   576→  let acc = 0
   577→  box(width: pw, height: 2 * r + 1mm, {
   578→    for (i, v) in values.enumerate() {
   579→      let a0 = acc / total * 360deg - 90deg
   580→      acc = acc + v
   581→      let a1 = acc / total * 360deg - 90deg
   582→      let steps = calc.max(2, calc.floor((a1 - a0) / 2.5deg))
   583→      for s in range(steps) {
   584→        let b0 = a0 + (a1 - a0) * s / steps
   585→        let b1 = a0 + (a1 - a0) * (s + 1) / steps
   586→        // tiny angular overlap hides hairline seams between fan triangles
   587→        let p0 = (cx + r * calc.cos(b0), cy + r * calc.sin(b0))
   588→        let p1 = (cx + r * calc.cos(b1 + 0.5deg), cy + r * calc.sin(b1 + 0.5deg))
   589→        place(top + left, curve(fill: colors.at(i), stroke: none,
   590→          curve.move((cx, cy)),
   591→          curve.line((p0.at(0), p0.at(1))),
   592→          curve.line((p1.at(0), p1.at(1))),
   593→          curve.close(),
   594→        ))
   595→      }
   596→    }
   597→    // wedge separators + outer rim, drawn on top of the fan
   598→    let acc2 = 0
   599→    for (i, v) in values.enumerate() {
   600→      let a = acc2 / total * 360deg - 90deg
   601→      acc2 = acc2 + v
   602→      place(top + left, dx: cx, dy: cy, line(angle: a, length: r, stroke: 0.8pt + paper))
   603→    }
